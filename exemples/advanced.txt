// 5. POO complète
classe Forme {
  constructeur(couleur) {
    this.couleur = couleur;
  }
  
  decrire() {
    retourne "Forme " + this.couleur;
  }
}

classe Cercle etend Forme {
  constructeur(couleur, rayon) {
    super(couleur);
    this.rayon = rayon;
  }
  
  aire() {
    retourne Math.PI * this.rayon * this.rayon;
  }
}
