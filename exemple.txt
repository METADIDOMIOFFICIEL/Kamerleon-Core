// Fonction avec paramètres par défaut
def saluer(nom = "Invité") {
  retourne "Bonjour " + nom;
}

// Fonction fléchée
ndem multiplier = (x, y) => x * y;

// Rest parameters
def somme(...nombres) {
  ndem total = 0;
  pour (ndem n de nombres) {
    total += n;
  }
  retourne total;
}

yamo(saluer("Alice"));
yamo("Multiplication: " + multiplier(4, 5));
yamo("Somme: " + somme(1, 2, 3, 4, 5));

// 5. Programmation orientée objet
yamo("\n5. PROGRAMMATION ORIENTÉE OBJET");

classe Animal {
  constructeur(nom) {
    voici.nom = nom;
  }
  
  parler() {
    yamo(voici.nom + " fait un bruit");
  }
}

classe Chien etend Animal {
  constructeur(nom, race) {
    super(nom);
    voici.race = race;
  }
  
  parler() {
    yamo(voici.nom + " aboie");
  }
  
  decrire() {
    retourne voici.nom + " est un " + voici.race;
  }
}

ndem monChien = nouveau Chien("Rex", "Berger Allemand");
monChien.parler();
yamo(monChien.decrire());

def attendre(delai) {
  retourne nouvelle Promesse((resoudre) => {
    setTimeout(resoudre, delai);
  });
}

async def tacheAsynchrone() {
  yamo("Début de la tâche asynchrone");
  attendre 1000;
  yamo("Tâche terminée après 1 seconde");
}
